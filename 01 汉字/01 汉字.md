# 汉字

> Chinese Character Index

## Characters

```dataviewjs
const pages = Array.from(
    dv.pages('"中文/01 汉字"')
        .where(p => p.type === "汉字" && p.pinyin)
);

// Convert Pinyin with tone marks to normal letters.
// Examples:
// ài   -> ai
// hǎo  -> hao
// qíng -> qing
function normalizePinyin(text) {
    return String(text)
        .normalize("NFD")
        .replace(/[\u0300-\u036f]/g, "")
        .toLowerCase();
}

// Group characters by the first letter of their Pinyin.
const groups = {};

for (const page of pages) {
    const normalized = normalizePinyin(page.pinyin);
    const initial = normalized.charAt(0).toUpperCase();

    if (!groups[initial]) {
        groups[initial] = [];
    }

    groups[initial].push(page);
}

// Display groups alphabetically.
for (const initial of Object.keys(groups).sort()) {
    dv.header(3, initial);

    const pagesInGroup = groups[initial];

    // Sort characters by Pinyin inside each group.
    pagesInGroup.sort((a, b) =>
        normalizePinyin(a.pinyin)
            .localeCompare(normalizePinyin(b.pinyin))
    );

    dv.table(
        ["汉字", "Pinyin", "Meaning", "Status"],
        pagesInGroup.map(p => [
            p.file.link,
            p.pinyin ?? "",
            p.meaning ?? "",
            p.status ?? ""
        ])
    );
}
```

## Recently Learning

```dataview
TABLE WITHOUT ID
    file.link AS "汉字",
    pinyin AS "Pinyin",
    meaning AS "Meaning"
FROM "中文/01 汉字"
WHERE type = "汉字"
AND status = "learning"
LIMIT 5
SORT pinyin ASC
```
